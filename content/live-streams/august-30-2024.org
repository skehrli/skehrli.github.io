#+begin_src scheme

  (def get-command-grammar
    '{:target (<- (some (range "az")))
      :get "get "
      :main (* :get :target)})

  (defn process-command []
    (def input (file/read stdin :line))
    (print "The command was: " input)
    (pp (peg/match get-command-grammar input))

    # Recursive loop!
    (process-command))

  (defn main
    [& args]
    (process-command))

#+end_src
